    Logging¶

    All SAOS modules share one logging queue and one listener thread per process. Scripts create the logger with LoggingHelper(level).logger and pass it to every module (logger=...); a module created without a logger uses get_logger(), which returns the same shared root logger.

    import logging
from SAOS.LoggingHelper import LoggingHelper
from SAOS.Telescope import Telescope

log = LoggingHelper(logging.INFO)
tel = Telescope(diameter=1.5, resolution=64, logger=log.logger)

    def get_logger(logging_level=None):
    """
    Root logger fed through the shared queue and listener of SAOS (console output, "time - level - message").

    Parameters
    ----------
    logging_level : int, optional
        Minimum level of the root logger. None keeps the current one (WARNING when the listener is created here).

    Returns
    -------
    logging.Logger
        The root logger.
    """
    global _listener
    root_logger = logging.getLogger()
    with _lock:
        if _listener is None:
            log_queue = Queue()
            console_handler = logging.StreamHandler()
            console_handler.setFormatter(logging.Formatter("%(asctime)s - %(levelname)s - %(message)s"))
            _listener = logging.handlers.QueueListener(log_queue, console_handler)
            root_logger.addHandler(logging.handlers.QueueHandler(log_queue))
            root_logger.setLevel(logging.WARNING if logging_level is None else logging_level)
            atexit.register(_stop_listener)
        elif logging_level is not None:
            root_logger.setLevel(logging_level)
        if _listener._thread is None:                     # not started yet, or stopped by the user: (re)start it
            _listener.start()
    return root_logger
